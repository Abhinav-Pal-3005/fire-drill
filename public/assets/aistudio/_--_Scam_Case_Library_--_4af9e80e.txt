  function toggleSwitch(button) {
    const isChecked = button.getAttribute('aria-checked') === 'true';
    const nextState = !isChecked;
    button.setAttribute('aria-checked', nextState.toString());

    const knob = button.querySelector('.toggle-knob');
    if (nextState) {
      button.classList.remove('bg-surface-dim');
      button.classList.add('bg-primary-container');
      knob.classList.remove('translate-x-0');
      knob.classList.add('translate-x-5');
      showToast('Preference enabled');
    } else {
      button.classList.remove('bg-primary-container');
      button.classList.add('bg-surface-dim');
      knob.classList.remove('translate-x-5');
      knob.classList.add('translate-x-0');
      showToast('Preference disabled');
    }
  }

  function showResetDialog() {
    const modal = document.getElementById('reset-modal');
    modal.classList.remove('hidden');
  }

  function hideResetDialog() {
    const modal = document.getElementById('reset-modal');
    modal.classList.add('hidden');
  }

  function confirmReset() {
    hideResetDialog();

    msg.textContent = text;
    toast.classList.remove('opacity-0', 'pointer-events-none');

    setTimeout(() => {
      toast.classList.remove('opacity-100');
      toast.classList.add('opacity-0', 'pointer-events-none');
    }, 2400);
  }
